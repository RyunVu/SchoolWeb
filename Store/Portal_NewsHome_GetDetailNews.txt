CREATE PROCEDURE [dbo].[Portal_NewsHome_GetDetailNews]
@p_lang nvarchar(500) = NULL,
@p_unitcode nvarchar(500) = NULL,
@p_alias nvarchar(500) = NULL,
@p_isrefreshed int = 0,
@p_page_index int = 1,
@p_page_size int = 10
AS
BEGIN
	
	IF(@p_isrefreshed = 1)
	BEGIN
		UPDATE News SET CountView = CountView + 1
		WHERE  
		Status = 1
		AND
		(@p_unitcode IS NULL
			OR UnitCode = @p_unitcode)
		AND
		(@p_alias IS NULL
			OR Alias = @p_alias);
	END
	
	DECLARE @p_code nvarchar(500) = NULL;

	SET @p_code = (SELECT Code FROM News n
	WHERE  
	n.Status = 1
	AND
	(@p_unitcode IS NULL
		OR n.UnitCode = @p_unitcode)
	AND
	(@p_alias IS NULL
		OR n.Alias = @p_alias));
	
    -- SET NOCOUNT ON added to prevent extra result sets from
    -- interfering with SELECT statements.
    SET NOCOUNT ON;
   
	SELECT n.Id, n.NewTypeId, n.Code, n.Alias, 
	case when @p_lang = 'en' then ISNULL(n.Title_En, n.Title) else n.Title end Title, 
	case when @p_lang = 'en' then ISNULL(n.Content_En, n.Content) else n.Content end Content, 
	n.ImageUrl, n.ShortContent, n.[Order], n.CountView, n.Tag, n.Status, n.Description, n.CreateDate, n.CreateUserId,
	n.UpdateDate, n.UpdateUserId, n.LanguageId, n.UnitCode,
	n.Title_En, n.Content_En, n.ShortContent_En,
	n.AudioUrl, n.AudioCreateDate, 
	case when @p_lang = 'en' then ISNULL(gc.Name_En, gc.Name) else gc.Name end CodeName
	FROM News n
	left join GeneralCategories gc on n.NewTypeId = gc.Id
	WHERE  
	n.Status = 1
	AND
	(@p_unitcode IS NULL
		OR n.UnitCode = @p_unitcode)
	AND
	(@p_alias IS NULL
		OR n.Alias = @p_alias);
   
   	SELECT n.Id, n.NewTypeId, n.Code, n.Alias, 
	case when @p_lang = 'en' then ISNULL(n.Title_En, n.Title) else n.Title end Title, 
	case when @p_lang = 'en' then ISNULL(n.Content_En, n.Content) else n.Content end Content, 
	n.ImageUrl, n.ShortContent, n.[Order], n.CountView, n.Tag, n.Status, n.Description, n.CreateDate, n.CreateUserId,
	n.UpdateDate, n.UpdateUserId, n.LanguageId, n.UnitCode,
	n.Title_En, n.Content_En, n.ShortContent_En,
	n.AudioUrl, n.AudioCreateDate FROM News n
	WHERE  
	n.Status = 1
	AND
	(@p_code IS NULL
		OR n.Code = @p_code)
	AND
	(@p_unitcode IS NULL
		OR n.UnitCode = @p_unitcode)
	AND
	(@p_alias IS NULL
		OR n.Alias != @p_alias)
		
	ORDER BY n.CreateDate DESC
        
    OFFSET ((@p_page_index - 1) * @p_page_size) ROWS FETCH NEXT @p_page_size ROWS ONLY;
END
